void SetFreqTimer1(float freq) \{\
    T1CONbits.TCKPS = 0b00;\
    if (FCY / freq > 65535) \{\
        T1CONbits.TCKPS = 0b01;\
        if (FCY / freq / 8 > 65535) \{\
            T1CONbits.TCKPS = 0b10;\
            if (FCY / freq / 64 > 65535) \{\
                T1CONbits.TCKPS = 0b11;\
                PR1 = (int) (FCY / freq / 256);\
            \} else\
                PR1 = (int) (FCY / freq / 64);\
        \} else\
            PR1 = (int) (FCY / freq / 8);\
    \} else\
        PR1 = (int) (FCY / freq);\
\}\
